def sha(path): return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def td(tensor):
    t = tensor.detach().cpu().contiguous()
    return hashlib.sha256(f'{t.dtype}:{tuple(t.shape)}:'.encode() + t.numpy().tobytes()).hexdigest()
def load(path): return json.loads(Path(path).read_text())
def treehash(tree):
    h = hashlib.sha256()
    def rec(value):
        if torch.is_tensor(value):
            t = value.detach().cpu().contiguous()
            h.update(str((str(t.dtype), tuple(t.shape))).encode()); h.update(t.numpy().tobytes())
        elif isinstance(value, dict):
            for k in sorted(value, key=str): h.update(str(k).encode()); rec(value[k])
        elif isinstance(value, (list, tuple)):
            for v in value: rec(v)
        else: h.update(repr(value).encode())
    rec(tree); return h.hexdigest()
def independently_predict(head, fields):
    # Algebraically reconstruct production outputs, do not invoke fixed-geometry wrapper.
    outputs=[]
    with torch.no_grad():
        for f in fields.split(16):
            n,k,d,h,w=f.shape
            x=head.trunk(f.reshape(n,k*d,h,w)); hidden=x.shape[1]//k
            common=head.value(x).reshape(n,k,head.out_channels,h,w).mean(1)
            delta=head.contrast(x.reshape(n,k,hidden,h,w).mean(1))
            outputs.append(torch.stack([common-delta, common+delta],1))
    return torch.cat(outputs).numpy()
def metrics(pred, target):
    a=((pred[:,1]-pred[:,0])/2).astype(np.float64)
    t=((target[:,1]-target[:,0])/2).astype(np.float64)
    cue=np.sum(t*t,axis=(1,2,3))>1e-20
    aa=a[cue].ravel(); tt=t[cue].ravel(); power=np.dot(tt,tt); cross=np.dot(aa,tt)
    return {'contrast_nmse':float(np.dot(aa-tt,aa-tt)/power), 'cosine':float(cross/np.sqrt(np.dot(aa,aa)*power)), 'projection_gain':float(cross/power)}
def choices(pred, cache):
    pred=pred.astype(np.float64)
    residual=pred-cache['observed'].numpy().astype(np.float64)[:,None]
    costs=np.sum(residual**2/(2*cache['total_variance'].numpy().astype(np.float64)[:,None]),axis=(2,3,4))
    totals={}
    for i,idn in enumerate(cache['identities']):
        if idn.get('world') in ('0','1'):
            totals.setdefault(idn['seed'],[0.,0.])[int(idn['world'])]+=float(costs[i,1]-costs[i,0])
    return sum(a>0 and b<0 for a,b in totals.values()),len(totals)
