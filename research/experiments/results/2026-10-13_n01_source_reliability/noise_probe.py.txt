"""Post-hoc analytic-estimator diagnostic; no network/optimization/test tuning."""
import json
from pathlib import Path
import torch
from owac.core.encoder_contract import MeasurementFrame
from owac.frameworks.n01.config import N01Config
from owac.frameworks.n01.state import init_state,push
from owac.frameworks.n01.support import SupportDomainIntegrator

torch.set_num_threads(1)
base=N01Config(input_mode='bayer_cfa',channels=4,planes=4,spatial_radii=(0,),neighbourhood_radius=0,time_depths=(8,),learnable_gate=False).validate()
conditions={'static':(.3,.3),'weak_jump':(.38,.38),'strong_jump':(1.,1.),'current_outlier':(1.,.3)}
rows=[]
for seed in [99771,99772,99773]:
 noise=torch.randn(8,4096,4,1,1,generator=torch.Generator().manual_seed(seed))*.08
 for name,(new_mean,truth) in conditions.items():
  state=init_state(base,4096,{'camera_ids':['camera'],'shape':(1,1)})
  for i in range(8):
   x=noise[i]+(.3 if i<7 else new_mean)
   state=push(state,MeasurementFrame(values=x,valid_mask=torch.ones_like(x,dtype=torch.bool),camera_id='camera',timestamp=torch.full((4096,),float(i)),frame_ids=torch.full((4096,),i),time_unit='frame',metadata={'variance':torch.full_like(x,.08**2)}))
  record={'seed':seed,'condition':name,'independent_cells':16384,'current_mse':float(((state.values[0]-truth)**2).mean())}
  for mode in ['none','innovation_risk']:
   integrator=SupportDomainIntegrator(base.replace(source_reliability=mode))
   stats,_=integrator.domain_statistics(state,torch.ones(4096,4,2,1,1))
   record[mode]={'mse':float(((stats.mu-truth)**2).mean()),'variance_proxy':float(stats.variance_proxy.mean()),'age':float(stats.age.mean()),'history_mass':float(integrator.last_history_mass.mean()) if mode!='none' else .875}
  rows.append(record)
  print(json.dumps(record),flush=True)
result={'diagnostic':'post-hoc source estimator, independent synthetic scalar CFA cells; no learned model; no hyperparameter selection','noise_sigma':.08,'history_length':8,'old_mean':.3,'rows':rows}
Path(__file__).with_name('noise-probe.json').write_text(json.dumps(result,indent=2)+'\n')
